cmake_minimum_required(VERSION 3.21)
project(MVeditor VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC OFF)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

option(MV_BUILD_TESTS "Build the engine test-suite" ON)
option(MV_BUILD_APP   "Build the GUI application" ON)

find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Multimedia MultimediaWidgets)
if(MV_BUILD_TESTS)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
endif()

file(GLOB CORE_SRC CONFIGURE_DEPENDS src/core/*.cpp src/core/*.h)
add_library(mveditor_core STATIC ${CORE_SRC})
target_include_directories(mveditor_core PUBLIC src/core)
target_link_libraries(mveditor_core PUBLIC Qt6::Core Qt6::Gui)
if(MSVC)
  target_compile_options(mveditor_core PUBLIC /utf-8 /W3)
else()
  target_compile_options(mveditor_core PUBLIC -Wall -Wextra -Wno-unused-parameter)
endif()

if(MV_BUILD_TESTS)
  enable_testing()
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE mveditor_core Qt6::Core)
  add_test(NAME engine COMMAND engine_tests)
endif()

if(MV_BUILD_APP)
  file(GLOB UI_SRC CONFIGURE_DEPENDS src/ui/*.cpp src/ui/*.h)
  set(APP_RES resources/resources.qrc)
  if(WIN32)
    list(APPEND APP_RES resources/app.rc)
  endif()
  add_executable(mveditor WIN32 src/main.cpp ${UI_SRC} ${APP_RES})
  target_include_directories(mveditor PRIVATE src/ui)
  target_link_libraries(mveditor PRIVATE mveditor_core Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
  set_target_properties(mveditor PROPERTIES OUTPUT_NAME "M Veditor")
  target_compile_definitions(mveditor PRIVATE MV_VERSION="${PROJECT_VERSION}")
  if(WIN32)
    target_link_libraries(mveditor PRIVATE winmm)
  endif()
  install(TARGETS mveditor RUNTIME DESTINATION .)
  if(MV_BUILD_TESTS)
    add_executable(gui_smoke tests/gui_smoke.cpp ${UI_SRC} resources/resources.qrc)
    target_include_directories(gui_smoke PRIVATE src/ui)
    target_link_libraries(gui_smoke PRIVATE mveditor_core Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Test)
    target_compile_definitions(gui_smoke PRIVATE MV_VERSION="${PROJECT_VERSION}")
    add_test(NAME gui COMMAND gui_smoke)
  endif()
endif()
